const SHEET_NAME = "items"; const FOLDER_NAME = "WarehouseImages"; function doGet() { return json({ ok:true, items:getItems() }); } function doPost(e) { const data = JSON.parse(e.postData.contents || "{}"); if(data.action === "save"){ saveItem(data.item); return json({ ok:true }); } return json({ ok:false }); } function saveItem(item){ const ss = SpreadsheetApp.getActive(); const sh = ss.getSheetByName(SHEET_NAME) || ss.insertSheet(SHEET_NAME); if(sh.getLastRow()===0){ sh.appendRow(["name","category","shelf","side","level","zone","notes","image_url","date"]); } let imgUrl = ""; if(item.image){ const folder = getFolder(); const blob = Utilities.newBlob( Utilities.base64Decode(item.image.split(",")[1]), "image/jpeg", "img.jpg" ); const file = folder.createFile(blob); imgUrl = file.getUrl(); } sh.appendRow([ item.name, item.category, item.shelf, item.side, item.level, item.zone, item.notes, imgUrl, new Date() ]); } function getItems(){ const sh = SpreadsheetApp.getActive().getSheetByName(SHEET_NAME); if(!sh) return []; const rows = sh.getDataRange().getValues(); rows.shift(); return rows.map(r=>({ name:r[0], category:r[1], shelf:r[2], side:r[3], level:r[4], zone:r[5], notes:r[6], image_url:r[7] })); } function getFolder(){ const fs = DriveApp.getFoldersByName(FOLDER_NAME); return fs.hasNext()?fs.next():DriveApp.createFolder(FOLDER_NAME); } function json(o){ return ContentService.createTextOutput(JSON.stringify(o)) .setMimeType(ContentService.MimeType.JSON); }
